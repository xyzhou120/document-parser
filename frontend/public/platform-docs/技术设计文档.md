## 三、数据库设计

### 3.1 Phase 0核心表（4张）

#### 表1: documents（文档表）

```sql
-- 文档表：存储上传的文档基本信息
CREATE TABLE documents (
    document_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    
    -- 文件信息
    file_name VARCHAR(500) NOT NULL,
    original_name VARCHAR(500) NOT NULL,  -- 用户上传的原始文件名
    file_path TEXT NOT NULL,              -- MinIO存储路径
    file_size BIGINT NOT NULL CHECK (file_size > 0),
    file_hash VARCHAR(64) NOT NULL,       -- SHA256，用于去重
    mime_type VARCHAR(100) NOT NULL,
    page_count INTEGER,
    
    -- 业务元数据（Phase 0暂时为空，Phase 1从附件消息获取）
    mine_name VARCHAR(200),
    mine_id VARCHAR(100),
    checklist_type VARCHAR(100),
    report_period VARCHAR(50),
    
    -- 上传信息
    uploaded_by VARCHAR(200),
    uploaded_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    
    -- 解析状态
    parse_status VARCHAR(50) NOT NULL DEFAULT 'PENDING',
    -- PENDING: 待解析
    -- PARSING: 解析中
    -- COMPLETED: 已完成
    -- FAILED: 失败
    
    -- 时间戳
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 索引
CREATE INDEX idx_documents_hash ON documents(file_hash);
CREATE INDEX idx_documents_status ON documents(parse_status);
CREATE INDEX idx_documents_created ON documents(created_at DESC);
CREATE INDEX idx_documents_mine_checklist 
    ON documents(mine_id, checklist_type, report_period) 
    WHERE mine_id IS NOT NULL;

-- 触发器：自动更新updated_at
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER update_documents_updated_at
    BEFORE UPDATE ON documents
    FOR EACH ROW
    EXECUTE FUNCTION update_updated_at_column();
```

#### 表2: parse_runs（解析运行记录表）

```sql
-- 解析运行记录表：每次解析的完整记录
CREATE TABLE parse_runs (
    parse_run_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    document_id UUID NOT NULL REFERENCES documents(document_id) ON DELETE CASCADE,
    
    -- 运行状态（参考D组状态机）
    status VARCHAR(50) NOT NULL DEFAULT 'REQUESTED',
    -- REQUESTED: 已请求
    -- QUEUED: 已入队
    -- RUNNING: 运行中
    -- COMPLETED: 已完成
    -- FAILED: 失败
    
    -- 解析配置（Phase 0固定，Phase 1从配置读取）
    parser_profile VARCHAR(100) NOT NULL DEFAULT 'default',
    parser_version VARCHAR(20) NOT NULL DEFAULT 'v1.0',
    
    -- 性能指标
    started_at TIMESTAMPTZ,
    completed_at TIMESTAMPTZ,
    duration_seconds INTEGER,
    
    -- 错误信息
    error_code VARCHAR(100),
    error_message TEXT,
    error_trace TEXT,
    
    -- 统计信息
    pages_processed INTEGER,
    fields_extracted INTEGER,
    
    -- Celery任务ID
    celery_task_id VARCHAR(255),
    
    -- 时间戳
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 索引
CREATE INDEX idx_parse_runs_document ON parse_runs(document_id);
CREATE INDEX idx_parse_runs_status ON parse_runs(status);
CREATE INDEX idx_parse_runs_celery ON parse_runs(celery_task_id);
CREATE INDEX idx_parse_runs_created ON parse_runs(created_at DESC);

-- 触发器
CREATE TRIGGER update_parse_runs_updated_at
    BEFORE UPDATE ON parse_runs
    FOR EACH ROW
    EXECUTE FUNCTION update_updated_at_column();
```

#### 表3: extracted_fields（提取字段表）

```sql
-- 提取字段表：存储每次解析提取的字段结果
CREATE TABLE extracted_fields (
    field_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    parse_run_id UUID NOT NULL REFERENCES parse_runs(parse_run_id) ON DELETE CASCADE,
    document_id UUID NOT NULL REFERENCES documents(document_id) ON DELETE CASCADE,
    
    -- 字段信息
    field_key VARCHAR(100) NOT NULL,      -- 字段唯一标识，如 mine_name
    field_name VARCHAR(200),              -- 字段中文名称，如 "煤矿名称"
    field_type VARCHAR(50),               -- STRING/NUMBER/DATE/BOOLEAN
    
    -- 提取结果
    field_value TEXT,                     -- 原始值
    field_value_normalized JSONB,         -- 标准化值（数值、日期等）
    field_unit VARCHAR(50),               -- 单位（如适用）
    
    -- 提取元数据
    extraction_method VARCHAR(50),        -- keyword/regex/llm/vision/multipath
    confidence FLOAT CHECK (confidence >= 0 AND confidence <= 1),
    
    -- 坐标信息（D组Locator模型，JSONB存储）
    evidence JSONB,
    -- 格式：
    -- {
    --   "page_number": 12,
    --   "bbox": {"x0": 120.5, "y0": 340.2, "x1": 280.3, "y1": 360.8},
    --   "coordinate_system": "PDF_POINT",
    --   "quote": "原文引用",
    --   "page_image_url": "http://minio/bucket/doc-001/page-12.jpg"
    -- }
    
    -- 多实例处理（如字段在文档中出现多次）
    is_primary BOOLEAN DEFAULT TRUE,
    instance_index INTEGER DEFAULT 0,
    all_candidates JSONB,                 -- 所有候选结果
    
    -- 时间戳
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 索引
CREATE INDEX idx_extracted_fields_parse_run ON extracted_fields(parse_run_id);
CREATE INDEX idx_extracted_fields_document ON extracted_fields(document_id);
CREATE INDEX idx_extracted_fields_key ON extracted_fields(field_key);
CREATE INDEX idx_extracted_fields_doc_key 
    ON extracted_fields(document_id, field_key);
```

#### 表4: parse_logs（解析日志表）

```sql
-- 解析日志表：记录解析过程中的关键步骤和事件
CREATE TABLE parse_logs (
    log_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    parse_run_id UUID NOT NULL REFERENCES parse_runs(parse_run_id) ON DELETE CASCADE,
    
    -- 日志级别
    level VARCHAR(20) NOT NULL,  -- DEBUG/INFO/WARNING/ERROR
    
    -- 阶段信息
    stage VARCHAR(50),  -- pdf_parse/ocr/extraction/quality_check
    
    -- 日志内容
    message TEXT NOT NULL,
    details JSONB,  -- 详细信息（如性能数据、错误堆栈等）
    
    -- 时间戳
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 索引
CREATE INDEX idx_parse_logs_run ON parse_logs(parse_run_id, created_at);
CREATE INDEX idx_parse_logs_level ON parse_logs(level);
```

```python
class MultiPathExtractor:
    """
    多路径融合提取器
    
    Phase 0: keyword + llm
    Phase 1: keyword + llm + vision
    """
    
    def __init__(self):
        self.keyword_extractor = KeywordExtractor()
        self.llm_extractor = LLMExtractor()
    
    async def extract_field(
        self,
        document_content: str,
        field_config: Dict
    ) -> ExtractedField:
        """
        多路径提取单个字段
        
        1. 尝试规则提取（keyword）
        2. 如果置信度低，LLM兜底
        3. 返回最佳结果
        """
        candidates = []
        
        # 路径1: 规则提取
        keyword_result = await self.keyword_extractor.extract(
            document_content,
            field_config
        )
        if keyword_result:
            candidates.append(keyword_result)
        
        # 路径2: LLM提取（兜底）
        if not keyword_result or keyword_result.confidence < 0.8:
            llm_result = await self.llm_extractor.extract(
                document_content,
                field_config
            )
            if llm_result:
                candidates.append(llm_result)
        
        # 选择最佳结果
        if not candidates:
            return None
        
        return max(candidates, key=lambda x: x.confidence)
```

```python
# app/config.py

from pydantic_settings import BaseSettings, SettingsConfigDict
from pydantic import Field
from typing import List, Dict, Optional
import yaml
from pathlib import Path

class DatabaseSettings(BaseSettings):
    host: str = "localhost"
    port: int = 5432
    name: str = "document_parser"
    user: str = "postgres"
    password: str = "postgres"
    pool_size: int = 10
    max_overflow: int = 20
    
    @property
    def url(self) -> str:
        return f"postgresql+asyncpg://{self.user}:{self.password}@{self.host}:{self.port}/{self.name}"

class RedisSettings(BaseSettings):
    host: str = "localhost"
    port: int = 6379
    db: int = 0
    password: Optional[str] = None
    max_connections: int = 50
    
    @property
    def url(self) -> str:
        auth = f":{self.password}@" if self.password else ""
        return f"redis://{auth}{self.host}:{self.port}/{self.db}"

class MinIOSettings(BaseSettings):
    endpoint: str = "localhost:9000"
    access_key: str = "minioadmin"
    secret_key: str = "minioadmin"
    bucket: str = "documents"
    secure: bool = False

class ModelSettings(BaseSettings):
    base_url: str
    api_key: Optional[str] = None
    model_name: str
    timeout: int = 60
    max_retries: int = 3

class Settings(BaseSettings):
    # 应用配置
    app_name: str = "document-parser"
    app_version: str = "1.0.0"
    debug: bool = False
    host: str = "0.0.0.0"
    port: int = 8000
    
    # 子配置
    database: DatabaseSettings = Field(default_factory=DatabaseSettings)
    redis: RedisSettings = Field(default_factory=RedisSettings)
    minio: MinIOSettings = Field(default_factory=MinIOSettings)
    
    # 模型配置
    text_llm: ModelSettings
    vision_llm: ModelSettings
    ocr: ModelSettings
    
    # 日志
    log_level: str = "INFO"
    
    # 上传
    max_file_size: int = 104857600  # 100MB
    allowed_extensions: List[str] = [".pdf"]
    
    # Phase 0字段配置
    fields_config: Dict = {}
    
    model_config = SettingsConfigDict(
        env_file=".env",
        env_nested_delimiter="__",
        case_sensitive=False
    )
    
    @classmethod
    def load_from_yaml(cls, env: str = "development"):
        """从YAML加载配置"""
        config_dir = Path(__file__).parent.parent / "config"
        
        # 加载default
        default_path = config_dir / "default.yaml"
        with open(default_path) as f:
            config = yaml.safe_load(f)
        
        # 加载环境特定配置
        env_path = config_dir / f"{env}.yaml"
        if env_path.exists():
            with open(env_path) as f:
                env_config = yaml.safe_load(f)
                config.update(env_config)
        
        # 加载local覆盖
        local_path = config_dir / "local.yaml"
        if local_path.exists():
            with open(local_path) as f:
                local_config = yaml.safe_load(f)
                config.update(local_config)
        
        return cls(**config)

# 全局配置实例
```
